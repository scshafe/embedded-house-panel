cmake_minimum_required(VERSION 3.20)

project(beaglebone_house_panel VERSION 0.1.0 LANGUAGES CXX)

set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)

option(BHP_BUILD_TESTS "Build host-testable domain tests" ON)
option(BHP_BUILD_PANEL_SUPERVISOR "Build the panel supervisor domain" ON)
option(BHP_BUILD_PANEL_UI "Build the neutral panel UI shell" ON)
option(BHP_BUILD_VOCABULARY_PANEL "Build the vocabulary panel domain" ON)
option(BHP_BUILD_MEDIA_BROKER "Build the media broker domain" ON)
option(BHP_BUILD_MEDIA_ALSA "Build the libasound capture source and its check tool" OFF)
option(BHP_BUILD_HTTP_LEXINET_CLIENT "Build libcurl/nlohmann Lexinet adapter" OFF)
option(BHP_BUILD_LVGL_RENDERER "Build the pinned LVGL target renderer" OFF)
option(BHP_BUILD_PANEL_RENDERER "Build shared LVGL surfaces and headless PNG renderer" OFF)
option(BHP_BUILD_PANEL_SIMULATOR "Build the native SDL2 virtual panel" OFF)
option(BHP_BUILD_PANEL_DRM "Build the explicit Linux simulation DRM backend" OFF)

if(BHP_BUILD_PANEL_SIMULATOR OR BHP_BUILD_PANEL_DRM)
  set(BHP_BUILD_PANEL_RENDERER ON)
endif()
if(BHP_BUILD_PANEL_RENDERER)
  set(BHP_BUILD_LVGL_RENDERER ON)
  include(cmake/lvgl.cmake)
endif()
include(cmake/nlohmann_json.cmake)

if(BHP_BUILD_TESTS)
  include(CTest)
  enable_testing()
endif()

if(BHP_BUILD_PANEL_SUPERVISOR OR BHP_BUILD_PANEL_UI)
  add_subdirectory(shared/application_identity)
  add_subdirectory(shared/panel_mode_protocol)
  add_subdirectory(shared/touch_observation_protocol)
endif()
add_subdirectory(shared/structured_logging)
add_subdirectory(shared/panel_fonts)
if(BHP_BUILD_HTTP_LEXINET_CLIENT)
  add_subdirectory(shared/http_runtime)
endif()
if(BHP_BUILD_PANEL_SUPERVISOR)
  add_subdirectory(panel_supervisor)
endif()
if(BHP_BUILD_PANEL_UI)
  add_subdirectory(panel_ui)
endif()
if(BHP_BUILD_VOCABULARY_PANEL)
  add_subdirectory(vocabulary_panel)
endif()
if(BHP_BUILD_MEDIA_BROKER)
  add_subdirectory(media_broker)
endif()
if(BHP_BUILD_PANEL_RENDERER)
  add_subdirectory(simulator)
endif()
