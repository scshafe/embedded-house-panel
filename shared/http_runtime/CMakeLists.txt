find_package(CURL REQUIRED)
find_package(Threads REQUIRED)
add_library(bhp_http_runtime STATIC src/http_runtime.cpp)
target_include_directories(bhp_http_runtime PUBLIC include)
target_compile_features(bhp_http_runtime PUBLIC cxx_std_17)
target_link_libraries(bhp_http_runtime PUBLIC CURL::libcurl Threads::Threads)
if(CMAKE_CXX_COMPILER_ID MATCHES "Clang|GNU")
 target_compile_options(bhp_http_runtime PRIVATE -Wall -Wextra -Wpedantic -Werror)
endif()

 target_link_libraries(http_runtime_test PRIVATE bhp_http_runtime)
 add_test(NAME http_runtime_test COMMAND http_runtime_test)
endif()
