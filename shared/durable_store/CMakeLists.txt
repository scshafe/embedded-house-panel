# @canonical durable_store_library -- shared outbox mechanics (issue #23).
add_library(bhp_durable_store STATIC src/durable_file.cpp src/backoff_and_upload.cpp)
add_library(bhp::durable_store ALIAS bhp_durable_store)
target_include_directories(bhp_durable_store PUBLIC include)
target_compile_features(bhp_durable_store PUBLIC cxx_std_17)
if(CMAKE_CXX_COMPILER_ID MATCHES "Clang|GNU")
  target_compile_options(bhp_durable_store PRIVATE -Wall -Wextra -Wpedantic -Werror)
endif()

if(BHP_BUILD_TESTS)
  foreach(test_name durable_store backoff resumable_upload)
    add_executable(${test_name}_test tests/${test_name}_test.cpp)
    target_link_libraries(${test_name}_test PRIVATE bhp_durable_store)
    if(CMAKE_CXX_COMPILER_ID MATCHES "Clang|GNU")
      target_compile_options(${test_name}_test PRIVATE -Wall -Wextra -Wpedantic -Werror)
    endif()
    add_test(NAME ${test_name}_test COMMAND ${test_name}_test)
  endforeach()
endif()
